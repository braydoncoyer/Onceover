import { SectionList } from '@/ui/lib/list/section-list';
import { StatusCard } from '@/components/StatusCard';
import { EmptyState } from '@/components/EmptyState';
import { useRideHistory } from './useRideHistory';
import styles from './RideHistoryPage.module.css';

interface RideHistoryPageProps {
  riderId: string;
}

export function RideHistoryPage({ riderId }: RideHistoryPageProps) {
  const { sections, emptyMessage, dimmedIds, dismiss } = useRideHistory(riderId);

  if (sections.value.length === 0) {
    return <EmptyState message={emptyMessage.value} />;
  }

  return (
    <SectionList
      className={styles.list}
      sections={sections.value}
      renderSectionHeader={(section) => (
        <h2 className={styles.sectionTitle}>{section.title}</h2>
      )}
      renderItem={(ride) => (
        <StatusCard
          ride={ride}
          style={{ opacity: dimmedIds.value.has(ride.id) ? 0.48 : 1 }}
          onDismiss={() => dismiss(ride.id)}
        />
      )}
    />
  );
}
