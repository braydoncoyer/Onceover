import type { RideStatus } from '@/types/rides';

export const RIDE_STATUS_LABELS: Record<RideStatus, string> = {
  requested: 'Requested',
  'driver-assigned': 'Driver assigned',
  'in-progress': 'On the way',
  completed: 'Completed',
  'cancelled-by-driver': 'Cancelled by driver',
  'cancelled-by-rider': 'Cancelled',
};

export function isDimmed(status: RideStatus): boolean {
  return status === 'cancelled-by-driver';
}
